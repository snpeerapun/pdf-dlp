cmake_minimum_required(VERSION 3.21)
project(pdfdlp_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Concurrent)
# PDFium prebuilt (bblanchon/pdfium-binaries): set PDFIUM_DIR to the extracted folder
set(PDFIUM_DIR "" CACHE PATH "pdfium prebuilt dir")
add_executable(pdfdlp_native MACOSX_BUNDLE main.cpp viewer.cpp viewer.h)
target_include_directories(pdfdlp_native PRIVATE "${PDFIUM_DIR}/include")
if(APPLE)
  target_link_libraries(pdfdlp_native PRIVATE "${PDFIUM_DIR}/lib/libpdfium.dylib")
  set_target_properties(pdfdlp_native PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "PDF DLP Native" BUILD_RPATH "${PDFIUM_DIR}/lib" INSTALL_RPATH "@executable_path/../Frameworks")
elseif(WIN32)
  target_link_libraries(pdfdlp_native PRIVATE "${PDFIUM_DIR}/lib/pdfium.dll.lib")
else()
  target_link_libraries(pdfdlp_native PRIVATE "${PDFIUM_DIR}/lib/libpdfium.so")
endif()
target_link_libraries(pdfdlp_native PRIVATE Qt6::Widgets Qt6::Concurrent)
