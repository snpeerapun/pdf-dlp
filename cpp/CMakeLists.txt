cmake_minimum_required(VERSION 3.21)
project(pdfdlp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Concurrent PrintSupport Network)
set(PDFIUM_DIR "${CMAKE_SOURCE_DIR}/pdfium" CACHE PATH "pdfium prebuilt dir")
set(PDFCORE_LIB "${CMAKE_SOURCE_DIR}/../core-ffi/target/release/libpdfcore.a" CACHE FILEPATH "Rust core static lib")
file(GLOB SRC src/*.cpp src/*.h)
if(WIN32)
  list(APPEND SRC installer/pdfdlp.rc)
endif()
add_executable(pdfdlp MACOSX_BUNDLE WIN32 ${SRC})
if(WIN32)
  target_compile_definitions(pdfdlp PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  target_compile_options(pdfdlp PRIVATE /utf-8 /EHsc)
endif()
target_include_directories(pdfdlp PRIVATE "${PDFIUM_DIR}/include" "${CMAKE_SOURCE_DIR}/../core-ffi")
target_link_libraries(pdfdlp PRIVATE Qt6::Widgets Qt6::Concurrent Qt6::PrintSupport Qt6::Network "${PDFCORE_LIB}")
if(APPLE)
  target_link_libraries(pdfdlp PRIVATE "${PDFIUM_DIR}/lib/libpdfium.dylib" "-framework CoreFoundation" "-framework Security")
  set_target_properties(pdfdlp PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "PDF DLP" MACOSX_BUNDLE_GUI_IDENTIFIER "co.th.oneable.pdfdlp" MACOSX_BUNDLE_BUNDLE_VERSION "0.1.0" MACOSX_BUNDLE_SHORT_VERSION_STRING "0.1.0" OUTPUT_NAME "PDF DLP")
elseif(WIN32)
  target_link_libraries(pdfdlp PRIVATE "${PDFIUM_DIR}/lib/pdfium.dll.lib" ws2_32 userenv bcrypt ntdll advapi32 ole32 shell32 RuntimeObject Propsys)
  set_target_properties(pdfdlp PROPERTIES OUTPUT_NAME "PDF DLP")
else()
  target_link_libraries(pdfdlp PRIVATE "${PDFIUM_DIR}/lib/libpdfium.so" pthread dl m)
endif()
